function openEventModal(dateStr, existingEvent = null) { const modal = document.getElementById("eventModal"); modal.style.display = "flex"; // Populate fields document.getElementById("modalDate").value = dateStr; document.getElementById("modalTitleInput").value = existingEvent?.title || ""; document.getElementById("modalLocationInput").value = existingEvent?.location || ""; document.getElementById("modalNotesInput").value = existingEvent?.notes || ""; // Save button document.getElementById("modalSaveBtn").onclick = () => { const newEvent = { date: document.getElementById("modalDate").value, title: document.getElementById("modalTitleInput").value, location: document.getElementById("modalLocationInput").value, class: existingEvent?.class || "User Event", notes: document.getElementById("modalNotesInput").value }; // Replace or add if (existingEvent) { const idx = userEvents.indexOf(existingEvent); if (idx !== -1) userEvents[idx] = newEvent; } else { userEvents.push(newEvent); } localStorage.setItem("userEvents", JSON.stringify(userEvents)); modal.style.display = "none"; renderCalendar(); }; // Cancel button document.getElementById("modalCancelBtn").onclick = () => { modal.style.display = "none"; }; }
